-- ============================================================ -- MIGRACION 23 — Arreglo de las fichas, desactivar y eliminar -- ============================================================ -- 1. EL ARREGLO URGENTE. La tabla personas no tiene un permiso -- de lectura general: lo tiene columna por columna, a -- proposito, para que el documento no se lea desde cualquier -- parte. La columna categoria_id que agrego la 21 nunca -- recibio el suyo, y como ficha_persona se lee con los -- permisos de quien entra, la vista entera se caia. Por eso -- no abria ninguna ficha, ni siquiera siendo administrador. -- -- Aqui se le da el permiso que faltaba. Y al final queda una -- consulta que lista las columnas de personas sin permiso, -- para que la proxima vez se vea antes de que falle. -- -- 2. DESACTIVAR Y ELIMINAR. Son dos cosas distintas: -- desactivar es para quien fue de la iglesia y se fue —se -- guarda todo y se puede volver—; eliminar es para un -- registro falso o un duplicado que no se dejo fusionar, y -- eso si se va de verdad. Lo que se borra queda anotado en -- una bitacora con quien lo borro y por que, porque un -- borrado sin rastro es un agujero. -- -- 3. LOS QUE NO SE PUDIERON MOVER A INVITADOS. La 21 movio a -- los visitantes sin datos, pero algunos se quedaron. Aqui -- se vuelve a intentar y, sobre todo, queda una vista que -- dice de cada uno POR QUE no se movio. -- -- Se aplica sobre la 22. -- ============================================================ -- ============================================================ -- 1. EL PERMISO QUE FALTABA -- ============================================================ grant select (categoria_id) on personas to authenticated; -- ============================================================ -- 2. DESACTIVAR, REACTIVAR, ELIMINAR -- ============================================================ -- La bitacora de lo borrado. No se pierde nada: su historial, sus -- cursos y sus notas siguen ahi, y el dia que vuelva se -- reactiva y sigue donde estaba. create or replace function desactivar_persona(p_persona uuid, p_motivo text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_yo uuid := mi_persona_id(); end if; select nombre, estado into v_nom, v_est from personas where id = p_persona; if v_nom is null then raise exception 'Esa persona no existe'; end if; if v_est = 'retirado' then raise exception '% ya está desactivada', v_nom; update persona_area set hasta = current_date where persona_id = p_persona and hasta is null; update seguimientos set estado = 'perdido', cerrado_en = now(), cerrado_por = v_yo, motivo_cierre = coalesce(nullif(trim(coalesce(p_motivo,'')), ''), 'Se desactivó la ficha') where persona_id = p_persona and estado = 'abierto'; insert into cambios (persona_id, asunto, detalle, accion, hecho_por) values (p_persona, 'datos', 'Ficha desactivada' || coalesce(': ' || nullif(trim(coalesce(p_motivo,'')), ''), ''), 'cambio', v_yo); end; $$; grant execute on function desactivar_persona(uuid, text) to authenticated; create or replace function reactivar_persona(p_persona uuid, p_lider uuid default null) returns void language plpgsql security definer set search_path = public as $$ declare v_yo uuid := mi_persona_id(); v_nom text; begin if v_yo is null then raise exception 'No hay sesión'; end if; if not (soy_admin() or ve_todo()) then raise exception 'Solo quien administra o dirige puede reactivar una ficha'; end if; Se pregunta ANTES de -- borrar para poder decirlo en pantalla, en vez de borrar y -- despues contar. create or replace function que_se_borra(p_persona uuid) returns table (que text, cuantos int, bloquea boolean) language sql stable security definer set search_path = public as $$ select * from ( values ('reuniones de cuidado', (select count(*)::int from contactos where persona_id = p_persona), false), ('asistencias', (select count(*)::int from asistencias where persona_id = p_persona), false), ('cursos y procesos', (select count(*)::int from inscripciones where persona_id = p_persona), false), ('áreas y grupos', (select count(*)::int from persona_area where persona_id = p_persona), false), ('notas de seguimiento', (select count(*)::int from notas_flujo where persona_id = p_persona), false), -- Estas si impiden el borrado: son cosas que esta persona le -- hizo a OTROS, y borrarla dejaria esas filas sin autor. Es para un registro falso o un duplicado -- que no se dejo fusionar, no para alguien que se fue: para eso -- esta desactivar. Si la persona dejo rastro en el trabajo de -- otros, no se borra y se dice exactamente que lo impide. create or replace function borrar_persona(p_persona uuid, p_motivo text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_yo uuid := mi_persona_id(); v_p personas%rowtype; v_bloqueo text; begin if v_yo is null then raise exception 'No hay sesión'; end if; if not soy_admin() then raise exception 'Solo quien administra puede eliminar una ficha'; end if; if p_persona = v_yo then raise exception 'No puedes eliminarte a ti mismo'; end if; select * into v_p from personas where id = p_persona; if v_p.id is null then raise exception 'Esa persona no existe'; end if; select string_agg(que || ' (' || cuantos || ')', ', ') into v_bloqueo from que_se_borra(p_persona) where bloquea; if v_bloqueo is not null then raise exception 'No se puede eliminar a %: dejó rastro en el trabajo de otros — %.
